# Funciones en Rust del módulo "matematicas" de la biblioteca estándar.
# Cada línea es: nombre | parámetros separados por comas (0 si no tiene) | descripción en español | ejemplo de una línea
raiz | numero | raíz cuadrada de un número (no negativo) | raiz(9)   # 3
potencia | base, exponente | eleva un número a otro | potencia(2, 10)   # 1024
absoluto | numero | valor sin signo | absoluto(-4)   # 4
redondear | numero | redondea al entero más cercano | redondear(2.5)   # 3
suelo | numero | redondea hacia abajo | suelo(2.7)   # 2
techo | numero | redondea hacia arriba | techo(2.1)   # 3
minimo | a, b | el menor de dos números | minimo(3, 8)   # 3
maximo | a, b | el mayor de dos números | maximo(3, 8)   # 8
aleatorio | 0 | número al azar entre 0 (incluido) y 1 (sin incluir) | aleatorio()
entero_al_azar | minimo, maximo | número entero al azar entre mínimo y máximo, los dos incluidos (como rand de PHP) | entero_al_azar(1, 6)   # un dado
